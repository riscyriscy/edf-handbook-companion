# Part 1: gdbserver, build host architecture
$ gcc -g -O0 -static -Wl,--build-id -fdebug-prefix-map=$PWD=. -o crash.full crash.c
$ objcopy --only-keep-debug crash.full symbols/.build-id/xx/yyyy.debug   # to the host
$ objcopy --strip-all crash.full dev/crash                                 # to the device
crash.full 787152 bytes
dev/crash 706576 bytes
symbols/.build-id/xx/yyyy.debug
